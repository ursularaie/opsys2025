*See ülesanne on lahendatud palju ai abiga. Vabandan, et see näeb selline välja ja siin ei ole kõike, mis vaja, aga loodan, et äkki selle eest saan midagi.<br>
```powershell
$env:HostIP = (
    Get-NetIPConfiguration |
    Where-Object {
        $_.IPv4DefaultGateway -ne $null -and
        $_.NetAdapter.Status -ne "Disconnected"
    }
).IPv4Address.IPAddress

& {
    Write-Output "Masina nimi: $env:computername PS Versioon: $($PSVersionTable.PSVersion)"
    Write-Output (systeminfo | findstr /B /C:"OS Name" /B /C:"OS Version")
    Write-Output "IPv4 aadress: $env:HostIP"
    Write-Output "V�rgumask: $((Get-WmiObject Win32_NetworkAdapterConfiguration | Where-Object {$_.IPEnabled -eq $true}).IPSubnet[0])"
    Write-Output "Vaikel��s: $((Get-NetRoute | where {$_.DestinationPrefix -eq '0.0.0.0/0'} | select -ExpandProperty NextHop))"
    Write-Output "DHCP luba:" 
    Get-NetIPInterface -AddressFamily IPv4 | Select-Object InterfaceAlias, InterfaceDescription, Dhcp
    Write-Output "MAC-aadress:" 
    Get-NetIPConfiguration | select @{n='MacAddress'; e={$_.NetAdapter.MacAddress}}
    Write-Output "Protsessor ja RAM:" 
    Get-CimInstance Win32_Processor | Select-Object @{n='CPU';e={$_.Name}}, @{n='RAM (GB)';e={[Math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB)}}

    $GPU = Get-CimInstance Win32_VideoController
    foreach ($Adapter in $GPU) {
        $DriverDate = $Adapter.DriverDate.ToShortDateString()
        $Resolution = "$($Adapter.CurrentHorizontalResolution) x $($Adapter.CurrentVerticalResolution)"
        Write-Output "--- GPU Information ---"
        Write-Output "Nimi:            $($Adapter.Name)"
        Write-Output "Draiveri versioon:  $($Adapter.DriverVersion)"
        Write-Output "Draiverir kuup�ev:      $DriverDate"
        Write-Output "Ekraani lahutus:      $Resolution"
    }

    $Disks = Get-Disk | Select-Object Number, FriendlyName, 
        @{Name="PartitionStyle"; Expression={$_.PartitionStyle}}, 
        @{Name="TotalSize_GB"; Expression={[Math]::Round($_.Size / 1GB, 2)}}

    $CDrive = Get-Volume -DriveLetter C | Select-Object DriveLetter, 
        @{Name="Total_GB"; Expression={[Math]::Round($_.Size / 1GB, 2)}}, 
        @{Name="Free_GB"; Expression={[Math]::Round($_.SizeRemaining / 1GB, 2)}},
        @{Name="PercentFree"; Expression={[Math]::Round(($_.SizeRemaining / $_.Size) * 100, 1)}}

    Write-Output "--- KETTA JA PARTITSIOONI TABELI INFO ---"
    $Disks | Format-Table -AutoSize
    Write-Output "--- C: DRAIV RUUM ---"
    $CDrive | Format-Table -AutoSize

    $TotalMachineGB = ($Disks | Measure-Object -Property TotalSize_GB -Sum).Sum
    Write-Output "mitu GB on arvuti kettad mahutavuselt: $TotalMachineGB GB"

    $AllAccounts = Get-CimInstance -ClassName Win32_UserAccount
    $AllAccounts | Select-Object `
        @{Name="AccountName"; Expression={$_.Name}},
        @{Name="Status"; Expression={if($_.Disabled -eq $True){"Disabled"}else{"Enabled"}}},
        @{Name="Type"; Expression={if($_.LocalAccount -eq $True){"Local"}else{"Domain"}}},
        Description | Format-Table -AutoSize

    Get-Process | Sort-Object WorkingSet64 -Descending | Select-Object `
        Id, 
        ProcessName, 
        @{Name="RAM (MB)"; Expression={[Math]::Round($_.WorkingSet64 / 1MB, 2)}}, 
        @{Name="CPU (s)"; Expression={[Math]::Round($_.CPU, 2)}},
        Description | Select-Object -First 20 | Format-Table -AutoSize

    Get-Process | Where-Object { $_.StartTime -ne $null } | 
        Sort-Object StartTime -Descending | 
        Select-Object -First 10 | 
        Select-Object Name, Id, StartTime | 
        Format-Table -AutoSize

    Write-Output (Get-Date -Format "dd.MM.yyyy HH:mm:ss")
} | Out-File -FilePath "ursula_raie.out.txt" -Encoding utf8
```
